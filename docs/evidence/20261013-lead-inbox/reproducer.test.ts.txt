import { randomUUID } from "node:crypto"
import { afterAll, afterEach, beforeAll, expect, it, vi } from "vitest"
const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
import prisma from "@/lib/prisma"
import { getLeadDashboard } from "@/actions/leads"
let memberId: string, otherCompanyId: string, opportunityId: string
beforeAll(async () => {
  session.companyId = (await prisma.company.create({ data: { name: "Fictional lead inbox diagnostic" } })).id
  otherCompanyId = (await prisma.company.create({ data: { name: "Fictional other lead inbox diagnostic" } })).id
  session.userId = (await prisma.user.create({ data: { email: `lead-inbox-diagnostic-${randomUUID()}@example.test` } })).id
  memberId = (await prisma.membership.create({ data: { companyId: session.companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })).id
  const client = await prisma.client.create({ data: { companyId: session.companyId, name: "Fictional inbox client" } })
  const pipeline = await prisma.pipeline.create({ data: { companyId: session.companyId, name: "Fictional inbox pipeline", stages: [{ id: "PROSPECT", title: "Prospect" }] } })
  opportunityId = (await prisma.opportunity.create({ data: { pipelineId: pipeline.id, clientId: client.id, title: "UNUSED-COMMERCIAL-INBOX-MARKER", status: "PROSPECT", valueCents: 987654 } })).id
  await prisma.leadCapture.createMany({ data: Array.from({ length: 151 }, (_, index) => ({
    companyId: session.companyId, clientId: client.id, opportunityId: index === 150 ? opportunityId : null,
    firstName: index === 0 ? "OLDEST-FICTIONAL-INBOX-LEAD" : `Fictional ${index}`, lastName: "Diagnostic",
    privacyAccepted: true, marketingOptIn: index === 0, fingerprint: `fictional-inbox-${index}`,
    createdAt: new Date(Date.UTC(2025, 0, 1, 0, index)),
  })) })
  await prisma.leadCapture.create({ data: { companyId: otherCompanyId, firstName: "OTHER-TENANT-INBOX-MARKER", lastName: "Diagnostic", privacyAccepted: true, fingerprint: "fictional-other-inbox" } })
})
afterEach(async () => { await prisma.membership.update({ where: { id: memberId }, data: { role: "OWNER" } }) })
afterAll(async () => {
  const companies = { companyId: { in: [session.companyId, otherCompanyId] } }
  await prisma.leadCapture.deleteMany({ where: companies })
  await prisma.opportunity.deleteMany({ where: { pipeline: companies } })
  await prisma.pipeline.deleteMany({ where: companies })
  await prisma.client.deleteMany({ where: companies })
  await prisma.company.deleteMany({ where: { id: { in: [session.companyId, otherCompanyId] } } })
  await prisma.user.deleteMany({ where: { id: session.userId } })
})
it("already excludes the other tenant and keeps the coherent client", async () => {
  const result = await getLeadDashboard()
  expect(JSON.stringify(result)).not.toContain("OTHER-TENANT-INBOX-MARKER")
  expect(result.leads[0].client?.name).toBe("Fictional inbox client")
})
it("counts all 151 coherent new leads rather than only the loaded 150", async () => {
  expect((await getLeadDashboard()).counts.NEW).toBe(151)
})
it("keeps the oldest coherent lead discoverable in the current locally searched inbox", async () => {
  expect((await getLeadDashboard()).leads.some(row => row.firstName === "OLDEST-FICTIONAL-INBOX-LEAD")).toBe(true)
})
it("does not send unused commercial data to a Technician with CRM but no Sales access", async () => {
  await prisma.membership.update({ where: { id: memberId }, data: { role: "TECHNICIAN" } })
  expect(JSON.stringify(await getLeadDashboard())).not.toContain("UNUSED-COMMERCIAL-INBOX-MARKER")
})
