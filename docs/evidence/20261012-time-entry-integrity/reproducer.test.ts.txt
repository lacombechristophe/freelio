import { randomUUID } from "node:crypto"
import { afterAll, afterEach, beforeAll, describe, expect, it, vi } from "vitest"
const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
import prisma from "@/lib/prisma"
import { getProjects } from "@/actions/projets"
import { createTimeEntry, deleteTimeEntry, getTimeEntries, updateTimeEntry } from "@/actions/temps"
describe.sequential("Project and time diagnostics on real SQL", () => {
  let clientId: string, projectId: string, invalidProjectId: string, foreignCompanyId: string
  beforeAll(async () => {
    session.companyId = (await prisma.company.create({ data: { name: "Fictional project and time diagnosis" } })).id
    foreignCompanyId = (await prisma.company.create({ data: { name: "Fictional foreign time diagnosis" } })).id
    session.userId = (await prisma.user.create({ data: { email: `time-diagnosis-${randomUUID()}@example.test` } })).id
    await prisma.membership.create({ data: { companyId: session.companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })
    clientId = (await prisma.client.create({ data: { companyId: session.companyId, name: "Fictional time customer" } })).id
    const foreign = await prisma.client.create({ data: { companyId: foreignCompanyId, name: "Fictional foreign time customer" } })
    projectId = (await prisma.project.create({ data: { companyId: session.companyId, clientId, name: "Older coherent project", createdAt: new Date("2000-01-01") } })).id
    invalidProjectId = (await prisma.project.create({ data: { companyId: session.companyId, clientId: foreign.id, name: "Incoherent time project" } })).id
  })
  afterEach(async () => {
    await prisma.timeEntry.deleteMany({ where: { project: { companyId: session.companyId } } })
    await prisma.invoice.deleteMany({ where: { companyId: session.companyId } })
    await prisma.project.deleteMany({ where: { companyId: session.companyId, id: { notIn: [projectId, invalidProjectId] } } })
    await prisma.project.updateMany({ where: { companyId: session.companyId }, data: { consumedCents: 0 } })
  })
  afterAll(async () => {
    await prisma.project.deleteMany({ where: { companyId: session.companyId } })
    await prisma.client.deleteMany({ where: { companyId: { in: [session.companyId, foreignCompanyId] } } })
    await prisma.company.deleteMany({ where: { id: { in: [session.companyId, foreignCompanyId] } } })
    await prisma.user.delete({ where: { id: session.userId } })
  })
  it("reproduces the default 20-project ceiling with a coherent twenty-first project", async () => {
    await prisma.project.createMany({ data: Array.from({ length: 20 }, (_, i) => ({ companyId: session.companyId, clientId, name: `Recent project ${i}`, createdAt: new Date("2001-01-01") })) })
    const rows = await getProjects()
    expect(rows).toHaveLength(20)
    expect(rows.some(row => row.id === projectId)).toBe(false)
  })
  it("reproduces 500 returned time entries out of 501 with the older entry absent", async () => {
    const older = await prisma.timeEntry.create({ data: { projectId, durationSec: 60, date: new Date("2000-01-01") } })
    await prisma.timeEntry.createMany({ data: Array.from({ length: 500 }, () => ({ projectId, durationSec: 60, date: new Date("2001-01-01") })) })
    const rows = await getTimeEntries()
    expect(rows).toHaveLength(500)
    expect(rows.some(row => row.id === older.id)).toBe(false)
    expect(await prisma.timeEntry.count({ where: { projectId } })).toBe(501)
  })
  it("reproduces creation of negative time and a negative consumed budget", async () => {
    const row = await createTimeEntry({ projectId, durationSec: -60 })
    expect(row.durationSec).toBe(-60)
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBeLessThan(0)
  })
  it("reproduces editing a duration to a negative value", async () => {
    const row = await prisma.timeEntry.create({ data: { projectId, durationSec: 60 } })
    expect((await updateTimeEntry(row.id, { durationSec: -60 })).durationSec).toBe(-60)
  })
  it("reproduces creation through a project whose client belongs to another company", async () => {
    const row = await createTimeEntry({ projectId: invalidProjectId, durationSec: 60 })
    expect(row.projectId).toBe(invalidProjectId)
  })
  it("reproduces editing a time entry with an incoherent project client", async () => {
    const row = await prisma.timeEntry.create({ data: { projectId: invalidProjectId, durationSec: 60 } })
    expect((await updateTimeEntry(row.id, { durationSec: 120 })).durationSec).toBe(120)
  })
  it("reproduces deletion of a time entry with an incoherent project client", async () => {
    const row = await prisma.timeEntry.create({ data: { projectId: invalidProjectId, durationSec: 60 } })
    await deleteTimeEntry(row.id)
    expect(await prisma.timeEntry.findUnique({ where: { id: row.id } })).toBeNull()
  })
  async function billed() {
    const invoice = await prisma.invoice.create({ data: { companyId: session.companyId, clientId, projectId, number: randomUUID(), object: "Fictional allocated invoice", status: "DRAFT", totalHtCents: 1000, totalTvaCents: 0, totalTtcCents: 1000, dueDate: new Date("2030-01-01") } })
    return prisma.timeEntry.create({ data: { projectId, durationSec: 60, invoiceId: invoice.id } })
  }
  it("reproduces editing a time entry already allocated to an invoice", async () => {
    const row = await billed()
    expect((await updateTimeEntry(row.id, { durationSec: 120 })).durationSec).toBe(120)
    expect((await prisma.timeEntry.findUniqueOrThrow({ where: { id: row.id } })).invoiceId).toBe(row.invoiceId)
  })
  it("reproduces deletion of a time entry already allocated to an invoice", async () => {
    const row = await billed()
    await deleteTimeEntry(row.id)
    expect(await prisma.timeEntry.findUnique({ where: { id: row.id } })).toBeNull()
    expect(await prisma.invoice.findUnique({ where: { id: row.invoiceId! } })).not.toBeNull()
  })
  it("keeps a coherent create, edit and delete path with a consistent final budget", async () => {
    const row = await createTimeEntry({ projectId, durationSec: 60 })
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBe(104)
    await updateTimeEntry(row.id, { durationSec: 120 })
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBe(208)
    await deleteTimeEntry(row.id)
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBe(0)
  })
})
