import { randomUUID } from "node:crypto"
import { afterAll, afterEach, beforeAll, describe, expect, it, vi } from "vitest"
const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
vi.mock("@/lib/finance/issued-invoice", () => ({ prepareIssuedInvoice: vi.fn(), discardIssuedInvoice: vi.fn() }))
import prisma from "@/lib/prisma"
import { createInvoiceFromTimeEntries } from "@/actions/factures"
import { updateTimeEntry } from "@/actions/temps"

describe.sequential("Time billing preflight diagnostics on real SQL", () => {
  let clientId: string, projectId: string, membershipId: string
  beforeAll(async () => {
    session.companyId = (await prisma.company.create({ data: { name: "Fictional time billing diagnosis", isTvaApplicable: false, invoicePrefix: "PREP-" } })).id
    session.userId = (await prisma.user.create({ data: { email: "billing-probe-" + randomUUID() + "@example.test" } })).id
    membershipId = (await prisma.membership.create({ data: { companyId: session.companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })).id
    const agency = await prisma.agency.create({ data: { companyId: session.companyId, name: "Fictional billing agency", code: "LOCAL" } })
    await prisma.agencyMembership.create({ data: { membershipId, agencyId: agency.id } })
    clientId = (await prisma.client.create({ data: { companyId: session.companyId, name: "Fictional billing customer" } })).id
    projectId = (await prisma.project.create({ data: { companyId: session.companyId, clientId, agencyId: agency.id, name: "Fictional billing project" } })).id
  })
  afterEach(async () => {
    vi.restoreAllMocks()
    await prisma.membership.update({ where: { id: membershipId }, data: { role: "OWNER" } })
    await prisma.timeEntry.deleteMany({ where: { projectId } })
    await prisma.invoice.deleteMany({ where: { companyId: session.companyId } })
    await prisma.auditLog.deleteMany({ where: { userId: session.userId } })
    await prisma.company.update({ where: { id: session.companyId }, data: { isTvaApplicable: false, invoicePrefix: "PREP-" } })
    await prisma.project.update({ where: { id: projectId }, data: { consumedCents: 0 } })
  })
  afterAll(async () => {
    await prisma.project.deleteMany({ where: { companyId: session.companyId } })
    await prisma.client.deleteMany({ where: { companyId: session.companyId } })
    await prisma.company.delete({ where: { id: session.companyId } })
    await prisma.user.delete({ where: { id: session.userId } })
  })
  const row = () => prisma.timeEntry.create({ data: { projectId, durationSec: 3600, description: "Fictional prepared text", date: new Date("2030-01-01") } })
  const invoiceInput = (id: string) => ({ timeEntryIds: [id], hourlyRateCents: 10000, dueDate: "2030-02-01", lineMode: "DETAIL" as const })
  function afterPreflight(change: () => Promise<unknown>) {
    const read = prisma.timeEntry.findMany.bind(prisma.timeEntry)
    vi.spyOn(prisma.timeEntry, "findMany").mockImplementationOnce((async args => { const entries = await read(args); await change(); return entries }) as typeof prisma.timeEntry.findMany)
  }
  it("reproduces billing a stale duration after a real edit", async () => {
    const entry = await row(); afterPreflight(async () => { expect(await updateTimeEntry(entry.id, { durationSec: 7200 })).toMatchObject({ ok: true }) })
    const invoice = await createInvoiceFromTimeEntries(invoiceInput(entry.id))
    expect(invoice.totalHtCents).toBe(10000)
    expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toMatchObject({ durationSec: 7200, invoiceId: invoice.id })
  })
  it("reproduces allocating a time made non-billable during preparation", async () => {
    const entry = await row(); afterPreflight(async () => { expect(await updateTimeEntry(entry.id, { isBillable: false })).toMatchObject({ ok: true }) })
    const invoice = await createInvoiceFromTimeEntries(invoiceInput(entry.id))
    expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toMatchObject({ isBillable: false, invoiceId: invoice.id })
  })
  it("reproduces freezing the old description after a real edit", async () => {
    const entry = await row(); afterPreflight(async () => { expect(await updateTimeEntry(entry.id, { description: "Fictional current text" })).toMatchObject({ ok: true }) })
    const invoice = await createInvoiceFromTimeEntries(invoiceInput(entry.id))
    expect(await prisma.invoiceLine.findFirstOrThrow({ where: { invoiceId: invoice.id } })).toMatchObject({ description: "Fictional prepared text" })
    expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toMatchObject({ description: "Fictional current text", invoiceId: invoice.id })
  })
  it("reproduces using the previous VAT setting", async () => {
    const entry = await row(); afterPreflight(() => prisma.company.update({ where: { id: session.companyId }, data: { isTvaApplicable: true } }))
    expect((await createInvoiceFromTimeEntries(invoiceInput(entry.id))).totalTvaCents).toBe(0)
    expect((await prisma.company.findUniqueOrThrow({ where: { id: session.companyId } })).isTvaApplicable).toBe(true)
  })
  it("reproduces using the previous invoice prefix", async () => {
    const entry = await row(); afterPreflight(() => prisma.company.update({ where: { id: session.companyId }, data: { invoicePrefix: "CURRENT-" } }))
    expect((await createInvoiceFromTimeEntries(invoiceInput(entry.id))).number).toMatch(/^PREP-/)
    expect((await prisma.company.findUniqueOrThrow({ where: { id: session.companyId } })).invoicePrefix).toBe("CURRENT-")
  })
  it("reproduces an Accounting denial when claiming an accessible time", async () => {
    const entry = await row(); await prisma.membership.update({ where: { id: membershipId }, data: { role: "ACCOUNTING" } })
    await expect(createInvoiceFromTimeEntries(invoiceInput(entry.id))).rejects.toThrow("FORBIDDEN:operations.write")
    expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toEqual(entry)
    expect(await prisma.invoice.count({ where: { companyId: session.companyId } })).toBe(0)
    expect(await prisma.auditLog.count({ where: { userId: session.userId, action: "CREATE_INVOICE_FROM_TIME" } })).toBe(0)
  })
  it("keeps coherent Owner invoicing as a positive control", async () => {
    const entry = await row(); const invoice = await createInvoiceFromTimeEntries(invoiceInput(entry.id))
    expect(invoice.totalHtCents).toBe(10000)
    expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toMatchObject({ invoiceId: invoice.id })
    expect(await prisma.auditLog.count({ where: { userId: session.userId, action: "CREATE_INVOICE_FROM_TIME" } })).toBe(1)
  })
})
