import { randomUUID } from "node:crypto"
import { afterAll, beforeAll, describe, expect, it, vi } from "vitest"
const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
import prisma from "@/lib/prisma"
import { createPortalToken, hashPortalToken, getPortalAccessFromToken } from "@/lib/portal/session"
import { revokeClientPortalAccess, updateClientPortalAppointment } from "@/actions/portal"
describe.sequential("Portal relationship diagnostic with real SQL and synthetic tenants", () => {
  let companyId: string, otherCompanyId: string, clientId: string, otherClientId: string, foreignContactId: string
  let incoherentAccessId: string, appointmentId: string
  const coherentToken = createPortalToken(), foreignClientToken = createPortalToken(), foreignContactToken = createPortalToken()
  beforeAll(async () => {
    companyId = session.companyId = (await prisma.company.create({ data: { name: "Fictional portal diagnostic" } })).id
    otherCompanyId = (await prisma.company.create({ data: { name: "Fictional other portal tenant" } })).id
    session.userId = (await prisma.user.create({ data: { email: `portal-diagnostic-${randomUUID()}@example.test` } })).id
    await prisma.membership.create({ data: { companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })
    clientId = (await prisma.client.create({ data: { companyId, name: "Fictional local portal client" } })).id
    otherClientId = (await prisma.client.create({ data: { companyId: otherCompanyId, name: "Fictional foreign portal client" } })).id
    foreignContactId = (await prisma.contact.create({ data: { clientId: otherClientId, firstName: "Fictional", lastName: "Foreign contact", email: "foreign-portal-diagnostic@example.test" } })).id
    const expiresAt = new Date(Date.now() + 3600000)
    await prisma.clientPortalAccess.create({ data: { companyId, clientId, tokenHash: hashPortalToken(coherentToken), expiresAt } })
    incoherentAccessId = (await prisma.clientPortalAccess.create({ data: { companyId, clientId: otherClientId, tokenHash: hashPortalToken(foreignClientToken), expiresAt } })).id
    await prisma.clientPortalAccess.create({ data: { companyId, clientId, contactId: foreignContactId, tokenHash: hashPortalToken(foreignContactToken), expiresAt } })
    appointmentId = (await prisma.clientPortalAppointmentRequest.create({ data: { companyId, clientId: otherClientId, subject: "Fictional incoherent request", preferredStart: new Date("2026-10-20T10:00:00Z") } })).id
  })
  afterAll(async () => {
    await prisma.auditLog.deleteMany({ where: { userId: session.userId } })
    await prisma.client.deleteMany({ where: { companyId: { in: [companyId, otherCompanyId] } } })
    await prisma.company.deleteMany({ where: { id: { in: [companyId, otherCompanyId] } } })
    await prisma.user.delete({ where: { id: session.userId } })
  })
  it("retains a coherent portal access", async () => {
    const access = await getPortalAccessFromToken(coherentToken)
    expect(access?.companyId).toBe(companyId); expect(access?.clientId).toBe(clientId)
  })
  it("rejects a live token linked to a foreign client", async () => {
    const access = await getPortalAccessFromToken(foreignClientToken)
    expect(access === null).toBe(true)
  })
  it("rejects a live token linked to a contact of another client", async () => {
    const access = await getPortalAccessFromToken(foreignContactToken)
    expect(access === null).toBe(true)
  })
  it("refuses an appointment with a foreign client before mutation or audit", async () => {
    let refused = false
    try { await updateClientPortalAppointment({ id: appointmentId, status: "CONFIRMED", response: "Fictional response" }) } catch { refused = true }
    const request = await prisma.clientPortalAppointmentRequest.findUniqueOrThrow({ where: { id: appointmentId } })
    const audits = await prisma.auditLog.count({ where: { userId: session.userId, action: "UPDATE_CLIENT_PORTAL_APPOINTMENT" } })
    expect({ refused, status: request.status, response: request.response, audits }).toEqual({ refused: true, status: "PENDING", response: null, audits: 0 })
  })
  it("refuses revocation of an access with a foreign client before mutation or audit", async () => {
    let refused = false
    try { await revokeClientPortalAccess(incoherentAccessId) } catch { refused = true }
    const access = await prisma.clientPortalAccess.findUniqueOrThrow({ where: { id: incoherentAccessId } })
    const audits = await prisma.auditLog.count({ where: { userId: session.userId, action: "REVOKE_CLIENT_PORTAL_ACCESS" } })
    expect({ refused, revoked: access.revokedAt !== null, audits }).toEqual({ refused: true, revoked: false, audits: 0 })
  })
})
