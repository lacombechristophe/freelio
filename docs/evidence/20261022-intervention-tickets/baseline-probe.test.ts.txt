/**
 * External baseline probe for the old New intervention Ticket SAV selector.
 * Baseline source: a699be73bae7104956e2102f6f3f9d587d8e2d96.
 * The current composer maps getOperationsDashboard().tickets directly. This
 * probe checks that payload, not a browser interaction or the future reader.
 * Expected baseline: 1 control passes, 2 product assertions fail.
 *
 * Scope after the fix: the dashboard cache may legitimately remain capped.
 * Keep this probe as baseline evidence. Test the new dedicated ticket choice
 * reader separately; do not call this unchanged probe a corrected-reader replay.
 * Uses only synthetic data in the explicitly selected fresh recipe database.
 */
import { randomUUID } from "node:crypto"
import { PrismaClient } from "@prisma/client"
import { afterAll, afterEach, beforeAll, describe, expect, it, vi } from "vitest"

const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
import { getOperationsDashboard } from "@/actions/operations"

const db = new PrismaClient()

describe.sequential("Baseline: New intervention ticket choices still use the bounded root cache", () => {
  let clientId: string, siteId: string, otherSiteId: string
  let firstTicketId: string, oldestEligibleTicketId: string
  const addedTicketIds: string[] = []

  // This deliberately reproduces the old composer's data.tickets.map source.
  // It must not be repointed to the future dedicated paginated choice reader.
  async function currentComposerChoices() { return (await getOperationsDashboard()).tickets }

  beforeAll(async () => {
    session.companyId = (await db.company.create({ data: { name: "Fictional intervention ticket baseline" } })).id
    session.userId = (await db.user.create({ data: { email: `ticket-baseline-${randomUUID()}@example.test`, companyId: session.companyId } })).id
    const member = await db.membership.create({ data: { companyId: session.companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })
    const agency = await db.agency.create({ data: { companyId: session.companyId, code: "BASELINE", name: "Fictional ticket choice agency" } })
    await db.agencyMembership.create({ data: { membershipId: member.id, agencyId: agency.id } })
    clientId = (await db.client.create({ data: { companyId: session.companyId, name: "Fictional intervention client" } })).id
    siteId = (await db.customerSite.create({ data: { companyId: session.companyId, clientId, agencyId: agency.id, label: "Fictional selected intervention site", address1: "Fictional address A" } })).id
    otherSiteId = (await db.customerSite.create({ data: { companyId: session.companyId, clientId, agencyId: agency.id, label: "Fictional other intervention site", address1: "Fictional address B" } })).id
    // One priority, unique timestamps: the root order deterministically omits
    // index 200. All 201 tickets satisfy the selected site's current eligibility.
    const newest = Date.parse("2030-01-02T12:00:00.000Z")
    await db.serviceTicket.createMany({ data: Array.from({ length: 201 }, (_, index) => ({
      companyId: session.companyId, clientId, siteId,
      number: `FICTIONAL-TICKET-${String(index).padStart(3, "0")}`,
      title: `Fictional eligible intervention ticket ${String(index).padStart(3, "0")}`,
      description: "Synthetic diagnostic; no real customer data.",
      priority: "NORMAL", status: "OPEN", updatedAt: new Date(newest - index * 1000),
    })) })
    firstTicketId = (await db.serviceTicket.findUniqueOrThrow({ where: { companyId_number: { companyId: session.companyId, number: "FICTIONAL-TICKET-000" } } })).id
    oldestEligibleTicketId = (await db.serviceTicket.findUniqueOrThrow({ where: { companyId_number: { companyId: session.companyId, number: "FICTIONAL-TICKET-200" } } })).id
  })

  afterEach(async () => {
    await db.serviceTicket.deleteMany({ where: { id: { in: addedTicketIds.splice(0) } } })
  })
  afterAll(async () => {
    try {
      if (session.companyId) {
        await db.serviceTicket.deleteMany({ where: { companyId: session.companyId } })
        await db.customerSite.deleteMany({ where: { companyId: session.companyId } })
        await db.client.deleteMany({ where: { companyId: session.companyId } })
        await db.auditLog.deleteMany({ where: { userId: session.userId } })
        await db.company.delete({ where: { id: session.companyId } })
      }
      if (session.userId) await db.user.delete({ where: { id: session.userId } })
    } finally { await db.$disconnect() }
  })

  it("keeps an initially visible coherent ticket in the current composer source", async () => {
    const options = await currentComposerChoices()
    expect(options).toHaveLength(200)
    expect(options.map(ticket => ticket.id)).toContain(firstTicketId)
    expect(await db.serviceTicket.count({ where: { companyId: session.companyId, clientId, siteId, status: "OPEN", mergedIntoTicketId: null } })).toBe(201)
  })

  it("makes the 201st eligible ticket reachable from the New intervention choice", async () => {
    expect(await db.serviceTicket.findUniqueOrThrow({ where: { id: oldestEligibleTicketId } })).toMatchObject({
      companyId: session.companyId, clientId, siteId, status: "OPEN", mergedIntoTicketId: null, equipmentId: null,
    })
    const options = await currentComposerChoices()
    expect(options).toHaveLength(200)
    // Product expectation: currently fails because the old source omits it.
    expect(options.map(ticket => ticket.id)).toContain(oldestEligibleTicketId)
  })

  it("excludes another site's ticket from the selected intervention site's choice", async () => {
    const incompatible = await db.serviceTicket.create({ data: {
      companyId: session.companyId, clientId, siteId: otherSiteId,
      number: "FICTIONAL-OTHER-SITE", title: "Fictional incompatible site ticket",
      description: "Same company, client and agency; only the site is incompatible.",
      priority: "URGENT", status: "OPEN", updatedAt: new Date("2030-01-03T12:00:00.000Z"),
    } })
    addedTicketIds.push(incompatible.id)
    expect(incompatible.siteId).not.toBe(siteId)
    const options = await currentComposerChoices()
    // Product expectation: currently fails because this source has no site filter.
    expect(options.map(ticket => ticket.id)).not.toContain(incompatible.id)
  })
})
