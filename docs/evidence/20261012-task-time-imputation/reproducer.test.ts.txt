import { randomUUID } from "node:crypto"
import { afterAll, afterEach, beforeAll, describe, expect, it, vi } from "vitest"
const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
vi.mock("@/lib/finance/issued-invoice", () => ({ prepareIssuedInvoice: vi.fn(), discardIssuedInvoice: vi.fn() }))
import prisma from "@/lib/prisma"
import { createTimeEntryFromOrganisationTask } from "@/actions/organisation"
import { createInvoiceFromTimeEntries } from "@/actions/factures"
import { updateTimeEntry } from "@/actions/temps"

describe.sequential("Time chain diagnostics on real SQL", () => {
  let clientId: string, projectId: string, invalidProjectId: string, foreignProjectId: string, foreignCompanyId: string
  beforeAll(async () => {
    session.companyId = (await prisma.company.create({ data: { name: "Fictional time chain diagnosis", isTvaApplicable: false } })).id
    foreignCompanyId = (await prisma.company.create({ data: { name: "Fictional foreign time chain" } })).id
    session.userId = (await prisma.user.create({ data: { email: `time-chain-${randomUUID()}@example.test` } })).id
    await prisma.membership.create({ data: { companyId: session.companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })
    clientId = (await prisma.client.create({ data: { companyId: session.companyId, name: "Fictional own customer" } })).id
    const foreign = await prisma.client.create({ data: { companyId: foreignCompanyId, name: "Fictional foreign customer" } })
    projectId = (await prisma.project.create({ data: { companyId: session.companyId, clientId, name: "Fictional coherent project" } })).id
    invalidProjectId = (await prisma.project.create({ data: { companyId: session.companyId, clientId: foreign.id, name: "Fictional inconsistent project" } })).id
    foreignProjectId = (await prisma.project.create({ data: { companyId: foreignCompanyId, clientId: foreign.id, name: "Fictional foreign project" } })).id
  })
  afterEach(async () => {
    vi.restoreAllMocks()
    await prisma.timeEntry.deleteMany({ where: { projectId: { in: [projectId, invalidProjectId, foreignProjectId] } } })
    await prisma.organisationTask.deleteMany({ where: { companyId: session.companyId } })
    await prisma.invoice.deleteMany({ where: { companyId: session.companyId } })
    await prisma.auditLog.deleteMany({ where: { userId: session.userId } })
    await prisma.project.updateMany({ where: { companyId: { in: [session.companyId, foreignCompanyId] } }, data: { consumedCents: 0 } })
  })
  afterAll(async () => {
    const where = { companyId: { in: [session.companyId, foreignCompanyId] } }
    await prisma.project.deleteMany({ where }); await prisma.client.deleteMany({ where })
    await prisma.company.deleteMany({ where: { id: { in: [session.companyId, foreignCompanyId] } } })
    await prisma.user.delete({ where: { id: session.userId } })
  })
  const task = (project = projectId) => prisma.organisationTask.create({ data: { companyId: session.companyId, projectId: project, title: "Fictional task", estimateMin: 1 } })
  it("reproduces a task creating time through a project whose client is foreign", async () => {
    const existing = await task(invalidProjectId)
    expect(await createTimeEntryFromOrganisationTask(existing.id, 1)).toEqual({ ok: true })
    expect(await prisma.timeEntry.count({ where: { projectId: invalidProjectId } })).toBe(1)
    expect((await prisma.organisationTask.findUniqueOrThrow({ where: { id: existing.id } })).status).toBe("DONE")
  })
  it("reproduces foreign-company time surviving a failed budget update", async () => {
    const existing = await task(foreignProjectId)
    await expect(createTimeEntryFromOrganisationTask(existing.id, 1)).rejects.toThrow()
    expect(await prisma.timeEntry.count({ where: { projectId: foreignProjectId } })).toBe(1)
    expect((await prisma.organisationTask.findUniqueOrThrow({ where: { id: existing.id } })).status).toBe("DONE")
  })
  it("reproduces a completed task and created time surviving a budget outage", async () => {
    const existing = await task()
    vi.spyOn(prisma.project, "update").mockRejectedValueOnce(new Error("Fictional budget outage"))
    await expect(createTimeEntryFromOrganisationTask(existing.id, 1)).rejects.toThrow("Fictional budget outage")
    expect(await prisma.timeEntry.count({ where: { projectId } })).toBe(1)
    expect((await prisma.organisationTask.findUniqueOrThrow({ where: { id: existing.id } })).status).toBe("DONE")
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBe(0)
  })
  it("reproduces billing a stale one-hour snapshot after a real two-hour edit", async () => {
    const entry = await prisma.timeEntry.create({ data: { projectId, durationSec: 3600 } })
    const read = prisma.timeEntry.findMany.bind(prisma.timeEntry)
    vi.spyOn(prisma.timeEntry, "findMany").mockImplementationOnce((async args => {
      const entries = await read(args)
      expect(await updateTimeEntry(entry.id, { durationSec: 7200 })).toMatchObject({ ok: true })
      return entries
    }) as typeof prisma.timeEntry.findMany)
    const invoice = await createInvoiceFromTimeEntries({ timeEntryIds: [entry.id], hourlyRateCents: 10000, dueDate: "2030-01-01" })
    expect(invoice.totalHtCents).toBe(10000)
    expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toMatchObject({ durationSec: 7200, invoiceId: invoice.id })
  })
  it("keeps a coherent task and current invoice as positive controls", async () => {
    const existing = await task()
    expect(await createTimeEntryFromOrganisationTask(existing.id, 60)).toEqual({ ok: true })
    const entries = await prisma.timeEntry.findMany({ where: { projectId } })
    expect(entries).toHaveLength(1)
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBe(6250)
    const invoice = await createInvoiceFromTimeEntries({ timeEntryIds: [entries[0].id], hourlyRateCents: 10000, dueDate: "2030-01-01" })
    expect(invoice.totalHtCents).toBe(10000)
  })
  it("reproduces two imputations of the same completed task", async () => {
    const existing = await task()
    await createTimeEntryFromOrganisationTask(existing.id, 1)
    await createTimeEntryFromOrganisationTask(existing.id, 1)
    expect(await prisma.timeEntry.count({ where: { projectId } })).toBe(2)
    expect((await prisma.project.findUniqueOrThrow({ where: { id: projectId } })).consumedCents).toBe(208)
  })
})
