import {randomUUID} from 'node:crypto'
import {afterAll,beforeAll,describe,expect,it,vi} from 'vitest'
const session=vi.hoisted(()=>({userId:'',companyId:''}))
vi.mock('server-only',()=>({}))
vi.mock('@/auth',()=>({auth:async()=>({user:{id:session.userId},companyId:session.companyId})}))
vi.mock('next/cache',()=>({revalidatePath:vi.fn()}))
import prisma from '@/lib/prisma'
import {getOperationsDashboard} from '@/actions/operations'
describe.sequential('Purchase form choice completeness diagnostic on isolated SQL',()=>{
 let productId:string,projectId:string
 beforeAll(async()=>{
  session.companyId=(await prisma.company.create({data:{name:'Fictional purchase choices diagnostic'}})).id
  session.userId=(await prisma.user.create({data:{email:`purchase-choices-${randomUUID()}@example.test`}})).id
  await prisma.membership.create({data:{companyId:session.companyId,userId:session.userId,role:'OWNER',status:'ACTIVE'}})
  const client=await prisma.client.create({data:{companyId:session.companyId,name:'Fictional purchase choice client'}})
  await prisma.product.createMany({data:Array.from({length:500},(_,i)=>({companyId:session.companyId,sku:`FAKE-${i}`,label:`Fictional product ${String(i).padStart(3,'0')}`}))})
  productId=(await prisma.product.create({data:{companyId:session.companyId,sku:'ZZ-LAST',label:'ZZ LAST PRODUCT',purchasePriceCents:789}})).id
  await prisma.project.createMany({data:Array.from({length:300},(_,i)=>({companyId:session.companyId,clientId:client.id,name:`Fictional project ${String(i).padStart(3,'0')}`}))})
  projectId=(await prisma.project.create({data:{companyId:session.companyId,clientId:client.id,name:'ZZ LAST PROJECT'}})).id
 })
 afterAll(async()=>{
  await prisma.project.deleteMany({where:{companyId:session.companyId}})
  await prisma.client.deleteMany({where:{companyId:session.companyId}})
  await prisma.company.delete({where:{id:session.companyId}})
  await prisma.user.delete({where:{id:session.userId}})
 })
 it('records 501 eligible products and 301 active projects',async()=>{
  expect(await prisma.product.count({where:{companyId:session.companyId,active:true}})).toBe(501)
  expect(await prisma.project.count({where:{companyId:session.companyId,status:'ACTIVE'}})).toBe(301)
 })
 it('returns the last product consumed by PurchaseOrderDialog',async()=>{
  expect((await getOperationsDashboard()).products.map(row=>row.id)).toContain(productId)
 })
 it('returns the last project consumed by PurchaseOrderDialog',async()=>{
  expect((await getOperationsDashboard()).projects.map(row=>row.id)).toContain(projectId)
 })
})
