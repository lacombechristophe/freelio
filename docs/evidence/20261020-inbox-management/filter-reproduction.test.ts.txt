import { afterAll, beforeAll, describe, expect, it, vi } from "vitest"
vi.mock("server-only", () => ({}))
import prisma from "@/lib/prisma"
import { requestContext } from "@/lib/context"
import { readInboxPage } from "@/lib/communications/inbox-reader"
let companyId: string, userId: string, channelId: string
beforeAll(async () => {
  companyId = (await prisma.company.create({ data: { name: "Fictional inbox filter baseline" } })).id
  userId = (await prisma.user.create({ data: { name: "Fictional inbox reader" } })).id
  channelId = (await prisma.communicationChannel.create({ data: { companyId, provider: "RESEND", emailAddress: "first@example.test", visibility: "SHARED" } })).id
  const second = await prisma.communicationChannel.create({ data: { companyId, provider: "GOOGLE", emailAddress: "second@example.test", visibility: "SHARED" } })
  await prisma.emailThread.createMany({ data: [
    { companyId, channelId, subject: "Own open case", assignedUserId: userId },
    { companyId, channelId, subject: "Free open case" },
    { companyId, channelId: second.id, subject: "Closed case", status: "CLOSED" },
  ] })
})
afterAll(async () => { await prisma.company.delete({ where: { id: companyId } }); await prisma.user.delete({ where: { id: userId } }) })
describe("inbox routing filter baseline on real SQL", () => {
  it.each(["mailbox", "assignment", "state"])("applies the %s filter instead of ignoring or rejecting it", async kind => {
    const input = kind === "mailbox" ? { channelId } : kind === "assignment" ? { assignment: "ME" } : { filter: "CLOSED" }
    const page = await requestContext.run({ userId, companyId, role: "SALES", membershipId: "fictional", agencyIds: null, actionPermission: "automation.read" }, () => readInboxPage(companyId, input as never))
    expect(page.total).toBe(kind === "mailbox" ? 2 : 1)
  })
})
