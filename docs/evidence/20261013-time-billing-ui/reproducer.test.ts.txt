import { randomUUID } from "node:crypto"
import { afterAll, afterEach, beforeAll, expect, it, vi } from "vitest"
const session = vi.hoisted(() => ({ userId: "", companyId: "" }))
vi.mock("server-only", () => ({}))
vi.mock("@/auth", () => ({ auth: async () => ({ user: { id: session.userId }, companyId: session.companyId }) }))
vi.mock("next/cache", () => ({ revalidatePath: vi.fn() }))
vi.mock("@/lib/finance/issued-invoice", () => ({ prepareIssuedInvoice: vi.fn(), discardIssuedInvoice: vi.fn() }))
import prisma from "@/lib/prisma"
import { createInvoiceFromTimeEntries, getUnbilledTimeEntries } from "@/actions/factures"
import { updateTimeEntry } from "@/actions/temps"
import { buildTimeInvoiceLines } from "@/lib/finance/time-invoice-lines"
import { calculateCommercialDocument } from "@/lib/finance/commercial-calculation"
let membership: string, first: string, second: string
beforeAll(async () => {
  session.companyId = (await prisma.company.create({ data: { name: "Fictional invoice UI diagnostic", isTvaApplicable: false } })).id
  session.userId = (await prisma.user.create({ data: { email: `invoice-ui-${randomUUID()}@example.test` } })).id
  membership = (await prisma.membership.create({ data: { companyId: session.companyId, userId: session.userId, role: "OWNER", status: "ACTIVE" } })).id
  const agency = await prisma.agency.create({ data: { companyId: session.companyId, code: "OWN", name: "Fictional assigned invoice agency" } })
  await prisma.agencyMembership.create({ data: { membershipId: membership, agencyId: agency.id } })
  const client = await prisma.client.create({ data: { companyId: session.companyId, name: "Fictional same invoice customer" } })
  first = (await prisma.project.create({ data: { companyId: session.companyId, clientId: client.id, agencyId: agency.id, name: "Fictional first invoice project" } })).id
  second = (await prisma.project.create({ data: { companyId: session.companyId, clientId: client.id, agencyId: agency.id, name: "Fictional second invoice project" } })).id
})
afterEach(async () => {
  await prisma.timeEntry.deleteMany({ where: { projectId: { in: [first, second] } } })
  await prisma.invoice.deleteMany({ where: { companyId: session.companyId } })
  await prisma.auditLog.deleteMany({ where: { userId: session.userId } })
  await prisma.membership.update({ where: { id: membership }, data: { role: "OWNER" } })
})
afterAll(async () => {
  await prisma.project.deleteMany({ where: { companyId: session.companyId } })
  await prisma.client.deleteMany({ where: { companyId: session.companyId } })
  await prisma.company.delete({ where: { id: session.companyId } })
  await prisma.user.delete({ where: { id: session.userId } })
})
const time = (projectId = first) => prisma.timeEntry.create({ data: { projectId, durationSec: 3600, date: new Date("2030-01-01"), description: "Fictional invoice UI time" } })
const input = (ids: string[]) => ({ timeEntryIds: ids, hourlyRateCents: 10000, dueDate: "2030-02-01", lineMode: "DETAIL" as const })
it("confirms an old 100 EUR preview produces a 200 EUR draft after an edit before submit", async () => {
  const entry = await time(), screen = await getUnbilledTimeEntries()
  if (!screen) throw new Error("Fictional invoice screen unavailable")
  const selected = screen.entries.filter(row => row.id === entry.id)
  expect(calculateCommercialDocument(buildTimeInvoiceLines(selected, 10000, 0, "DETAIL")).totalHtCents).toBe(10000)
  expect(await updateTimeEntry(entry.id, { durationSec: 7200 })).toMatchObject({ ok: true })
  const result = await createInvoiceFromTimeEntries(input(selected.map(row => row.id)))
  expect(result).toMatchObject({ ok: true })
  if (!result.ok) throw new Error(result.error)
  expect(await prisma.invoice.findUniqueOrThrow({ where: { id: result.value.id } })).toMatchObject({ status: "DRAFT", totalHtCents: 20000 })
  expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toMatchObject({ durationSec: 7200, invoiceId: result.value.id })
})
it("confirms Accounting can read both same-agency projects but cannot bill their times together", async () => {
  const entries = [await time(), await time(second)]
  await prisma.membership.update({ where: { id: membership }, data: { role: "ACCOUNTING" } })
  expect((await getUnbilledTimeEntries())?.entries).toHaveLength(2)
  await expect(createInvoiceFromTimeEntries(input(entries.map(row => row.id)))).rejects.toThrow("Facture introuvable")
  expect(await prisma.invoice.count({ where: { companyId: session.companyId } })).toBe(0)
  for (const entry of entries) expect(await prisma.timeEntry.findUniqueOrThrow({ where: { id: entry.id } })).toEqual(entry)
  expect(await prisma.auditLog.count({ where: { userId: session.userId, action: "CREATE_INVOICE_FROM_TIME" } })).toBe(0)
})
it("keeps the same two-project invoice available to Owner as a control", async () => {
  const entries = [await time(), await time(second)]
  const result = await createInvoiceFromTimeEntries(input(entries.map(row => row.id)))
  expect(result).toMatchObject({ ok: true })
  if (!result.ok) throw new Error(result.error)
  expect(await prisma.invoice.findUniqueOrThrow({ where: { id: result.value.id } })).toMatchObject({ projectId: null, totalHtCents: 20000 })
})
