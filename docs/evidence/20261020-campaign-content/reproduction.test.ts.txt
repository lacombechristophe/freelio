import { afterAll, describe, expect, it, vi } from "vitest"
vi.mock("server-only", () => ({}))
import prisma from "@/lib/prisma"
import { captureCampaignAudience } from "@/lib/marketing/campaign-activation"
import { renderEmailVariables } from "@/lib/automations/email"
const companies: string[] = [], users: string[] = []
afterAll(async () => {
  await prisma.auditLog.deleteMany({ where: { userId: { in: users } } })
  await prisma.company.deleteMany({ where: { id: { in: companies } } })
  await prisma.user.deleteMany({ where: { id: { in: users } } })
})
async function fixture(bodyHtml: string) {
  const company = await prisma.company.create({ data: { name: "Fictional content baseline" } }); companies.push(company.id)
  const user = await prisma.user.create({ data: { name: "Fictional baseline owner" } }); users.push(user.id)
  await prisma.membership.create({ data: { companyId: company.id, userId: user.id, role: "OWNER", status: "ACTIVE" } })
  const segment = await prisma.marketingSegment.create({ data: { companyId: company.id, name: "Fictional audience", kind: "STATIC", filters: {} } })
  const campaign = await prisma.marketingCampaign.create({ data: { companyId: company.id, segmentId: segment.id, name: "Fictional preflight", objective: "Recipe", channels: ["EMAIL"], status: "PLANNED" } })
  const channel = await prisma.communicationChannel.create({ data: { companyId: company.id, provider: "RESEND", status: "ACTIVE", visibility: "SHARED", emailAddress: "sender@example.test" } })
  const sequence = await prisma.emailSequence.create({ data: { companyId: company.id, campaignId: campaign.id, senderChannelId: channel.id, name: "Fictional sequence", status: "ACTIVE", steps: { create: { position: 0, type: "EMAIL", subject: "Fictional message", bodyHtml } } } })
  return { company, user, campaign, sequence }
}
describe("preflight regression before correction", () => {
  it.each([
    ["unknown variable", "<p>Bonjour {{contact.unknown}}</p>"],
    ["broken relative link", '<p>Réservez <a href="/booking">votre visite</a></p>'],
    ["empty sanitized body", "<script>discarded()</script><p>&nbsp; &#8203;</p>"],
  ])("rejects %s before capturing", async (_kind, bodyHtml) => {
    const f = await fixture(bodyHtml)
    await expect(captureCampaignAudience(f.company.id, f.user.id, { campaignId: f.campaign.id, sequenceId: f.sequence.id, version: 1 })).rejects.toThrow("Étape 1")
  })
  it("does not reinterpret a recipient value as a company variable", () => {
    expect(renderEmailVariables("{{contact.firstName}}", { company: { id: "fiction", name: "Fictional company", email: null }, lead: { id: "fiction", firstName: "{{company.name}}", lastName: "Recipient", email: null, projectType: null, city: null } })).toBe("{{company.name}}")
  })
})
